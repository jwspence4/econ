\begin{enumerate}

\item Suppose that a market with 10 consumers and 5 producers is perfectly competitive. Each consumer's individual demand is given by $q_D = 40 - 2P$. Each firm has fixed costs of \$50 and marginal costs of $MC = \frac{q_S}{4}$. Marginal cost is greater than or equal to average variable cost for all $q_S \geq 0$.

	\begin{enumerate}
	
	\item What is the short-run market equilibrium, and how much does each firm produce?
	
	\item What is each firm's profit at short-run equilibrium? Will firms enter or exit the market in the long run?
	
	\end{enumerate}

\item Newsprint (the paper used for newspapers) is produced in a perfectly competitive market. Each identical firm has a fixed cost of \$50, a variable cost of
	\begin{equation*}
	VC = 40q + \frac{q^2}{2}
	\end{equation*}
	and a marginal cost of
	\begin{equation*}
	MC = 40 + q.
	\end{equation*}
	
	\begin{enumerate}
	
	\item At what price will firms shut down in the short run?
	
	\item There are 12 identical firms in this industry. Current market demand for newsprint is
		\begin{equation*}
		Q_D = 360 - 2P.
		\end{equation*}
		What is the short-run equilibrium price?
		
	\end{enumerate}

\item The raspberry growing industry in the United States is perfectly competitive, and each producer has a marginal cost curve given by
	\begin{equation*}
	P = 20 + 2q,
	\end{equation*}
	and an average total cost curve given by
	\begin{equation*}
	P = 20 + q + \frac{144}{q}.
	\end{equation*}
	The market demand curve is
	\begin{equation*}
	Q_D = 2,488 - 2P.
	\end{equation*}
	
	\begin{enumerate}
	
	\item What is the long-run equilibrium price in this industry?
	
	\item How much does an individual firm produce at long-run equilibrium?
	
	\item How many active producers are in the raspberry growing industry in a long-run competitive equilibrium?
	
	\end{enumerate}

\end{enumerate}
