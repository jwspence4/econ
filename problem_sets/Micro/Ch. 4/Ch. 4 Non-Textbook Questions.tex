\begin{enumerate}

\item Assume that demand for coffee is given by $Q_D = 30 - 2P$, and supply is given by $Q_S = 4P$.

	\begin{enumerate}
	
	\item What is the equilibrium price of coffee?
	
	\item What is the equilibrium quantity of coffee?
	
	\end{enumerate}
	
\item Suppose that global demand for kombucha is $Q_D = 144 - 12P$, and global supply is $Q_S = 16 + 4P$.

	\begin{enumerate}
	
	\item What are the equilibrium price and quantity in the global kombucha market?
	
	\item Suppose domestic (i.e. not foreign) demand for kombucha is $Q_{D,Dom} = 80 - 8P$, and suppose foreign demand increases by 50\%. What is the new equilibrium price in the global kombucha market?
	
	\end{enumerate}

\end{enumerate}
